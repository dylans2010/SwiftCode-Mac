import Foundation

enum __FILENAME__ {
    case idle
    case loading(progress: Double)
    case success(data: Data)
    case failure(Error)

    var isTerminal: Bool {
        switch self {
        case .success, .failure:
            return true
        default:
            return false
        }
    }
}
